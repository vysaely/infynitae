struct R {
    ls: Lockstep,
    t: u64,
    w: Vec<usize>,
    e: Vec<ElementId>,
}

impl R {
    fn from(ls: Lockstep, n: usize) -> R {
        let mut ls = ls;
        ls.probe.auto_destroy_popups = true;
        let qh = ls.qh.clone();
        let labels = ["W1", "W2", "W3", "W4"];
        let mut w = Vec::new();
        for l in labels.iter().take(n) {
            w.push(ls.probe.toplevel(&qh, l, (400, 300), 0));
        }
        ls.pump();
        let e: Vec<ElementId> = ls.state.elements().collect();
        assert_eq!(e.len(), n);
        let mut r = R { ls, t: 10_000, w, e };
        r.adv();
        r
    }
    fn new(n: usize) -> R {
        R::from(Lockstep::new(Some(output((1920, 1080), 120))), n)
    }
    fn adv(&mut self) {
        self.ls.pump();
        let t = self.ls.now + Duration::from_micros(16_667);
        self.ls.run_until(t);
    }
    fn key(&mut self, k: u32, p: bool) {
        self.t += 1000;
        keyboard_key(&mut self.ls.state, &mut self.ls.input, k, p, self.t).expect("key");
        self.adv();
    }
    fn motion(&mut self, d: (f64, f64)) {
        self.t += 1000;
        pointer_motion(&mut self.ls.state, &mut self.ls.input, d, d, self.t).expect("motion");
        self.adv();
    }
    fn button(&mut self, b: u32, p: bool, m: infynitae_core::input::bindings::Modifiers) {
        self.t += 1000;
        pointer_button(&mut self.ls.state, &mut self.ls.input, b, p, m, self.t).expect("button");
        self.adv();
    }
    fn focus(&mut self, t: Option<ElementId>) {
        set_focus(&mut self.ls.state, &mut self.ls.input, t).expect("focus");
        self.adv();
    }
    fn target(&self) -> Option<ElementId> {
        self.ls.state.core().focus().target()
    }
    fn popup(&mut self, label: &'static str, parent: usize) -> usize {
        let qh = self.ls.qh.clone();
        let p = self.ls.probe.popup_with_grab(
            &qh,
            label,
            parent,
            PopupSpec { content: (120, 80), shadow: 0, anchor_rect: (10, 10, 1, 1), slide: false },
        );
        self.adv();
        p
    }
    fn viol(&self) -> Vec<String> {
        self.ls.probe.kb.violations.clone()
    }
    fn ptr_grabbed(&self) -> bool {
        self.ls.state.seat().get_pointer().unwrap().is_grabbed()
    }
    fn kb_grabbed(&self) -> bool {
        self.ls.state.keyboard_handle().is_grabbed()
    }
    fn label_of(&self, idx: usize) -> &'static str {
        self.ls.probe.surfaces[idx].label
    }
    /// Click-to-focus on the topmost window under the pointer; returns its (probe idx, element).
    fn click_focus(&mut self) -> (usize, ElementId) {
        self.motion((50.0, 50.0));
        self.button(BTN_LEFT, true, none());
        self.button(BTN_LEFT, false, none());
        let top = self.target().expect("click-to-focus");
        let i = self.e.iter().position(|&x| x == top).unwrap();
        (self.w[i], top)
    }
}

/// R01: a key event submitted with a FOREIGN port must fail closed — including the seat XKB
/// state (OBJ-27). Otherwise a rejected Shift becomes a modifier announced to clients on `enter`.
#[test]
fn r01_foreign_port_key_leaves_no_xkb_trace() {
    let mut r = R::new(2);
    r.focus(Some(r.e[0]));
    let mut other = Lockstep::new(Some(output((1920, 1080), 120)));
    let before = r.ls.state.keyboard_handle().modifier_state().serialized.depressed;
    let res = keyboard_key(&mut r.ls.state, &mut other.input, KEY_LEFTSHIFT, true, 1);
    assert!(res.is_err(), "foreign port accepted");
    let after = r.ls.state.keyboard_handle().modifier_state().serialized.depressed;
    r.focus(Some(r.e[1])); // enter + modifiers to W2
    println!("R01 depressed before={before:#x} after-rejected={after:#x}; client sees dep={:#x}", r.ls.probe.kb.depressed);
    assert_eq!(after, before, "REJECTED event mutated seat XKB state (no fail-closed)");
    assert_eq!(r.ls.probe.kb.depressed, 0, "client told a modifier is held that no accepted event pressed");
}

/// R15: causing (grab-serial) output A != current pointer output B != clock output C.
/// All cameras centred on the parent; parent visible on A, B, C. Popup: 600 wide at x=390 with
/// slide. Expected x by target: A(1920) 360, B(800) -200, C(1280) 40.
#[test]
fn r15_rule1_uses_the_causing_event_output() {
    let entries = ["A", "B", "C"]
        .iter()
        .map(|k| (infynitae_backend_headless::VirtualOutput::key(k), CanvasPos { x: 0, y: 0 }, 1.0))
        .collect();
    let ls = Lockstep::with(
        PersistedCameras::from_entries(entries).unwrap(),
        &[
            output_k("A", (1920, 1080), 120, 60_000),
            output_k("B", (800, 600), 120, 60_000),
            output_k("C", (1280, 720), 120, 144_000),
        ],
    );
    let mut r = R::from(ls, 1);
    r.motion((50.0, 50.0));
    r.button(BTN_LEFT, true, none()); // the causing event, on A
    let serial = r.ls.probe.last_serial;
    let out_click = r.ls.state.core().pointer_output();
    r.button(BTN_LEFT, false, none());
    assert_eq!(r.target(), Some(r.e[0]));
    r.motion((1000.0, 0.0)); // pointer now on B
    let out_now = r.ls.state.core().pointer_output();
    assert_ne!(out_click, out_now, "pointer did not change output");
    let qh = r.ls.qh.clone();
    let p = r.ls.probe.popup_with_grab_serial(
        &qh,
        "P",
        r.w[0],
        PopupSpec { content: (600, 50), shadow: 0, anchor_rect: (390, 10, 1, 1), slide: true },
        serial,
    );
    r.adv();
    let cfg = r.ls.probe.surfaces[p].popup_configure;
    println!("R15 click on {out_click:?}, pointer now {out_now:?}, popup configure={cfg:?} (A=360, B=-200, C=40)");
    assert_eq!(cfg.map(|c| c.0), Some(360), "popup constrained to the wrong output's viewport");

}

fn fds() -> usize {
    std::fs::read_dir("/proc/self/fd").map(|d| d.count()).unwrap_or(0)
}

/// R17: one long-lived Lockstep, 150 cycles: map window, click-focus, nested popup grabs with
/// distinct serials, CameraPan with popup, key cycles, focus transfer, zero-output + reconnect,
/// destroy window. Bounded: fds, focus history, key/button ownership tables, elements, grabs.
#[test]
fn r17_long_lived_stress_bounded_state() {
    let mut r = R::new(1);
    let mut samples = Vec::new();
    let mut cur = r.ls.output();
    for i in 0..150 {
        let qh = r.ls.qh.clone();
        let w = r.ls.probe.toplevel(&qh, "X", (300, 200), 0);
        r.adv();
        r.motion((1.0, 1.0));
        r.motion((-1.0, -1.0));
        r.button(BTN_LEFT, true, none());
        r.button(BTN_LEFT, false, none());
        let p1 = r.popup("P1", w);
        r.key(KEY_A, true);
        r.key(KEY_A, false);
        let s2 = r.ls.probe.last_serial;
        r.ls.probe.popup_with_grab_serial(&qh, "P2", p1, PopupSpec { content: (40, 30), shadow: 0, anchor_rect: (1, 1, 1, 1), slide: false }, s2);
        r.adv();
        r.key(KEY_LEFTMETA, true);
        r.button(BTN_MIDDLE, true, none());
        r.motion((3.0, 0.0));
        r.button(BTN_MIDDLE, false, none());
        r.key(KEY_LEFTMETA, true); // repeat
        r.key(KEY_LEFTMETA, false);
        r.key(KEY_LEFTMETA, true);
        r.key(KEY_ENTER, true); // focus_next
        r.key(KEY_ENTER, false);
        r.key(KEY_LEFTMETA, false);
        r.ls.disconnect(cur);
        r.adv();
        cur = r.ls.connect(&output((1920, 1080), 120));
        r.adv();
        r.ls.probe.destroy_toplevel(w);
        r.adv();
        if i == 9 || i == 149 {
            let core = r.ls.state.core();
            samples.push((
                fds(),
                core.focus().history().len(),
                core.keys().pressed().len(),
                core.buttons().pressed().len(),
                r.ls.state.elements().count(),
                r.ptr_grabbed(),
                r.kb_grabbed(),
            ));
        }
    }
    println!("R17 (fds, history, keys, buttons, elements, ptr_grabbed, kb_grabbed) @10 / @150: {samples:?}; viol={}", r.viol().len());
    let (a, b) = (samples[0], samples[1]);
    assert!(b.0 <= a.0 + 5 && b.1 <= a.1 && b.2 == 0 && b.3 == 0 && b.4 <= a.4, "unbounded state: {samples:?}");
}
