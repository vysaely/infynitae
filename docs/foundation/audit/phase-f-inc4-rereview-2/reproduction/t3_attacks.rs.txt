const KEY_K: u32 = 37;

struct R {
    ls: Lockstep,
    t: u64,
    w: Vec<usize>,
    e: Vec<ElementId>,
}

impl R {
    fn from(mut ls: Lockstep, n: usize, auto_destroy: bool) -> R {
        ls.probe.auto_destroy_popups = auto_destroy;
        let qh = ls.qh.clone();
        let labels = ["W1", "W2", "W3", "W4"];
        let mut w = Vec::new();
        for l in labels.iter().take(n) {
            w.push(ls.probe.toplevel(&qh, l, (400, 300), 0));
        }
        ls.pump();
        let e: Vec<ElementId> = ls.state.elements().collect();
        let mut r = R { ls, t: 10_000, w, e };
        r.adv();
        r
    }
    fn new(n: usize) -> R {
        R::from(Lockstep::new(Some(output((1920, 1080), 120))), n, true)
    }
    fn adv(&mut self) {
        self.ls.pump();
        let t = self.ls.now + Duration::from_micros(16_667);
        self.ls.run_until(t);
    }
    fn key(&mut self, k: u32, p: bool) {
        self.t += 1000;
        keyboard_key(&mut self.ls.state, &mut self.ls.input, k, p, self.t).expect("key");
        self.adv();
    }
    fn motion(&mut self, d: (f64, f64)) {
        self.t += 1000;
        pointer_motion(&mut self.ls.state, &mut self.ls.input, d, d, self.t).expect("motion");
        self.adv();
    }
    fn button(&mut self, b: u32, p: bool) {
        self.t += 1000;
        pointer_button(&mut self.ls.state, &mut self.ls.input, b, p, none(), self.t).expect("button");
        self.adv();
    }
    fn focus(&mut self, t: Option<ElementId>) {
        set_focus(&mut self.ls.state, &mut self.ls.input, t).expect("focus");
        self.adv();
    }
    fn target(&self) -> Option<ElementId> {
        self.ls.state.core().focus().target()
    }
    fn grab_popup(&mut self, label: &'static str, parent: usize, serial: u32) -> usize {
        let qh = self.ls.qh.clone();
        let p = self.ls.probe.popup_with_grab_serial(
            &qh,
            label,
            parent,
            PopupSpec { content: (120, 80), shadow: 0, anchor_rect: (10, 10, 1, 1), slide: false },
            serial,
        );
        self.adv();
        p
    }
    fn viol(&self) -> Vec<String> {
        self.ls.probe.kb.violations.clone()
    }
    fn ptr_grabbed(&self) -> bool {
        self.ls.state.seat().get_pointer().unwrap().is_grabbed()
    }
    fn kb_grabbed(&self) -> bool {
        self.ls.state.keyboard_handle().is_grabbed()
    }
    fn click_focus(&mut self) -> (usize, ElementId, u32) {
        self.motion((50.0, 50.0));
        self.button(BTN_LEFT, true);
        let s = self.ls.probe.last_serial;
        self.button(BTN_LEFT, false);
        let top = self.target().expect("click-to-focus");
        let i = self.e.iter().position(|&x| x == top).unwrap();
        (self.w[i], top, s)
    }
    fn root_of(&self, id: ElementId) -> Option<smithay::reexports::wayland_server::protocol::wl_surface::WlSurface> {
        self.ls.state.window(id).and_then(|w| w.toplevel().map(|t| t.wl_surface().clone()))
    }
    /// Coherence oracle: Core target vs Smithay keyboard focus vs client logical focus, then a
    /// probe key must be legally delivered (no wire violation) to the focused element's tree.
    fn coherent(&mut self, ctx: &str) {
        let tgt = self.target();
        let sf = self.ls.state.keyboard_handle().current_focus();
        let root = tgt.and_then(|t| self.root_of(t));
        let sf_alive = sf.as_ref().map(|s| s.is_alive());
        let client = self.ls.probe.kb.active.clone();
        let n = self.ls.probe.key_events.len();
        let v0 = self.viol().len();
        self.key(KEY_K, true);
        self.key(KEY_K, false);
        let got = self.ls.probe.key_events.len() - n;
        let newv: Vec<String> = self.viol()[v0..].to_vec();
        let ok = match tgt {
            Some(_) => sf_alive == Some(true) && client.is_some() && got == 2 && newv.is_empty()
                && (sf == root || client.as_deref().is_some_and(|c| c.starts_with('P'))),
            None => client.is_none() && got == 0 && newv.is_empty(),
        };
        assert!(
            ok,
            "[{ctx}] INCOHERENT: core={tgt:?} smithay_is_root={} smithay_alive={sf_alive:?} client={client:?} keys_delivered={got} new_violations={newv:?}",
            sf == root
        );
    }
    fn pan_press(&mut self) {
        self.key(KEY_LEFTMETA, true);
        self.button(BTN_MIDDLE, true);
    }
    fn pan_release(&mut self) {
        self.button(BTN_MIDDLE, false);
        self.key(KEY_LEFTMETA, false);
    }
    fn xkb(&self) -> (u32, u32, u32, usize) {
        let kb = self.ls.state.keyboard_handle();
        let m = kb.modifier_state().serialized;
        (m.depressed, m.latched, m.locked, kb.pressed_keys().len())
    }
}

    Lockstep::new(Some(output((1920, 1080), 120)))

#[test]
fn t02_rejected_events_leave_no_trace() {
    let mut r = R::new(2);
    r.focus(Some(r.e[0]));
    let mut f = foreign();
    let snap = |r: &R| (r.xkb(), r.ls.state.core().keys().pressed().to_vec(), r.ls.probe.kb_log.len());
    let cases: Vec<(&str, u32, bool, bool)> = vec![
        ("foreign Shift press", KEY_LEFTSHIFT, true, false),
        ("foreign Super press", KEY_LEFTMETA, true, false),
        ("foreign A press", KEY_A, true, false),
        ("foreign A release", KEY_A, false, false),
    ];
    for (name, k, p, _) in cases {
        let before = snap(&r);
        let res = keyboard_key(&mut r.ls.state, &mut f.input, k, p, 1);
        r.adv();
        assert!(res.is_err(), "{name}: accepted");
        assert_eq!(snap(&r), before, "{name}: rejected event left a trace");
    }
    // valid press -> foreign release (must not release) -> valid release
    r.key(KEY_LEFTSHIFT, true);
    let held = snap(&r);
    assert!(keyboard_key(&mut r.ls.state, &mut f.input, KEY_LEFTSHIFT, false, 1).is_err());
    r.adv();
    assert_eq!(snap(&r), held, "foreign release changed state of a valid press");
    r.key(KEY_LEFTSHIFT, false);
    // foreign press -> valid release (orphan): no client event
    let before = snap(&r);
    assert!(keyboard_key(&mut r.ls.state, &mut f.input, KEY_LEFTSHIFT, true, 1).is_err());
    r.key(KEY_LEFTSHIFT, false);
    let after = snap(&r);
    println!("T02 foreign press -> valid release: before={before:?} after={after:?} viol={:?}", r.viol());
    assert_eq!(after.0, before.0, "XKB changed");
    assert!(r.viol().is_empty(), "{:?}", r.viol());
}

fn t02b_token_is_bound_to_its_core() {

fn kb_seq(steps: &[(&str, u32, bool)]) -> R {
    let mut r = R::new(2);
    r.focus(Some(r.e[0]));
    for &(op, k, p) in steps {
        match op {
            "key" => r.key(k, p),
            "focus2" => r.focus(Some(r.e[1])),
            _ => unreachable!(),
        }
    }
    r
}

#[test]
fn t04_duplicate_presses() {
    let seqs: Vec<(&str, Vec<(&str, u32, bool)>)> = vec![
        ("dup", vec![("key", KEY_A, true), ("key", KEY_A, true), ("key", KEY_A, false)]),
        ("dup x2", vec![("key", KEY_A, true), ("key", KEY_A, true), ("key", KEY_A, true), ("key", KEY_A, false)]),
        ("dup after modifier", vec![("key", KEY_A, true), ("key", KEY_LEFTSHIFT, true), ("key", KEY_A, true), ("key", KEY_A, false), ("key", KEY_LEFTSHIFT, false)]),
        ("dup after focus", vec![("key", KEY_A, true), ("focus2", 0, false), ("key", KEY_A, true), ("key", KEY_A, false)]),
        ("dup modifier", vec![("key", KEY_LEFTSHIFT, true), ("key", KEY_LEFTSHIFT, true), ("key", KEY_LEFTSHIFT, false)]),
        ("dup Super", vec![("key", KEY_LEFTMETA, true), ("key", KEY_LEFTMETA, true), ("key", KEY_LEFTMETA, false)]),
    ];
    let mut fails = Vec::new();
    for (name, s) in seqs {
        let r = kb_seq(&s);
        let x = r.xkb();
        let c = (&r.ls.probe.kb.keys, r.ls.probe.kb.depressed);
        let ok = r.viol().is_empty() && x.0 == 0 && x.3 == 0 && c.0.is_empty() && c.1 == 0;
        println!("T04 {name}: viol={:?} seat(dep,lat,lock,pressed)={x:?} client keys={:?} dep={:#x}", r.viol(), c.0, c.1);
        if !ok {
            fails.push(name);
        }
    }
    assert!(fails.is_empty(), "duplicate-press sequences left inconsistent state: {fails:?}");
}

// ================================================================ INC4R-05 rule 1 provenance

fn three() -> R {
    let entries = ["A", "B", "C"]
        .iter()
        .map(|k| (infynitae_backend_headless::VirtualOutput::key(k), CanvasPos { x: 0, y: 0 }, 1.0))
        .collect();
    let ls = Lockstep::with(
        PersistedCameras::from_entries(entries).unwrap(),
        &[
            output_k("A", (1920, 1080), 120, 60_000),
            output_k("B", (800, 600), 120, 60_000),
            output_k("C", (1280, 720), 120, 144_000),
        ],
    );
    R::from(ls, 1, true)
}

fn slide_popup(r: &mut R, serial: u32) -> Option<i32> {
    let qh = r.ls.qh.clone();
    let p = r.ls.probe.popup_with_grab_serial(
        &qh,
        "P",
        r.w[0],
        PopupSpec { content: (600, 50), shadow: 0, anchor_rect: (390, 10, 1, 1), slide: true },
        serial,
    );
    r.adv();
    r.ls.probe.surfaces[p].popup_configure.map(|c| c.0)
}

fn out(r: &R) -> Option<OutputId> {
    r.ls.state.core().pointer_output()
}

/// Expected x: A 360, B -200, C 40.
#[test]
fn t05_rule1_variants() {
    let mut rows = Vec::new();
    // A != B != C (R15)
    {
        let mut r = three();
        let (_, _, s) = r.click_focus();
        r.motion((1000.0, 0.0));
        rows.push(("click A, pointer B, clock C", slide_popup(&mut r, s), Some(360)));
    }
    // A == B != C
    {
        let mut r = three();
        let (_, _, s) = r.click_focus();
        rows.push(("click A, pointer A", slide_popup(&mut r, s), Some(360)));
    }
    // A != B == C
    {
        let mut r = three();
        let (_, _, s) = r.click_focus();
        r.motion((1900.0, 0.0));
        let o = out(&r);
        rows.push(("click A, pointer C (clock)", slide_popup(&mut r, s), Some(360)));
        let _ = o;
    }
    // repeated crossings
    {
        let mut r = three();
        let (_, _, s) = r.click_focus();
        for _ in 0..3 {
            r.motion((1000.0, 0.0));
            r.motion((-1000.0, 0.0));
        }
        r.motion((1000.0, 0.0));
        rows.push(("click A, cross x7, pointer B", slide_popup(&mut r, s), Some(360)));
    }
    // a later click on B: serial of the EARLIER click on A must still mean A
    {
        let mut r = three();
        let (_, _, s) = r.click_focus();
        r.motion((1000.0, 0.0));
        r.button(BTN_LEFT, true);
        r.button(BTN_LEFT, false);
        rows.push(("click A (serial used), later click B", slide_popup(&mut r, s), Some(360)));
    }
    println!("T05 rows (got, want): {rows:?}");
    assert!(rows.iter().all(|(_, g, w)| g == w), "{rows:?}");
}

/// Provenance edge cases: printed for the dossier; asserted where the canon is unambiguous.
#[test]
fn t05b_rule1_edges() {
    // causing output removed before the popup request: rule 1 cannot apply -> rule 2 (clock C = 40)
    let mut r = three();
    let (_, _, s) = r.click_focus();
    let a = out(&r).unwrap();
    r.motion((1000.0, 0.0));
    r.ls.disconnect(a);
    r.adv();
    let removed = slide_popup(&mut r, s);
    // serial with no record (keyboard serial): fallback
    let mut r2 = three();
    r2.click_focus();
    r2.motion((1000.0, 0.0));
    r2.key(KEY_A, true);
    r2.key(KEY_A, false);
    let ks = r2.ls.probe.last_serial;

/// T99e: does the pointer lockout after nested-pan dismissal ever recover? 10 clicks, a keyboard
/// focus change (Super+Enter), a second window, and a zero-output cycle.
