def run_suite():
    cmd = "ulimit -n 1024; cargo test --locked --workspace --no-fail-fast 2>&1"
    p = subprocess.run(["bash", "-c", cmd], cwd=ROOT, capture_output=True, text=True)
    out = p.stdout + p.stderr
    failed = sorted(set(re.findall(r"^test (\S+) \.\.\. FAILED", out, re.M)))
    compile_err = "error[E" in out or "could not compile" in out
    return p.returncode, failed, compile_err, out

only = set(sys.argv[1:])
res = []
for mid, desc, f, old, new in M:
    if only and mid not in only:
        continue
    path = ROOT / f
    src = path.read_text()
    n = src.count(old)
    if n != 1:
        res.append((mid, desc, "BROKEN", f"pattern count {n}")); print(mid, "BROKEN pattern", n, flush=True); continue
    path.write_text(src.replace(old, new))
    try:
        rc, failed, cerr, out = run_suite()
    finally:
        path.write_text(src)
    if cerr:
        verdict, det = "BROKEN", "compile error: " + "; ".join(l for l in out.splitlines() if l.startswith("error"))[:300]
    elif failed:
        verdict, det = "DETECTED", ", ".join(failed)
    elif rc == 0:
        verdict, det = "MISSED", "suite green"
    else:
        verdict, det = "BROKEN", f"rc={rc} without failing test"
    res.append((mid, desc, verdict, det))
    print(f"{mid:3} {verdict:9} {desc} :: {det}", flush=True)

out = ROOT.parent / ("mutations" + ("_" + "_".join(sorted(only)) if only else "") + ".json")
out.write_text(json.dumps(res, indent=1))
subprocess.run(["git", "diff", "--stat"], cwd=ROOT)
